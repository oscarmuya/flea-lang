program     ::= statement* EOF
statement   ::= printStmt | makeStmt | exprStmt | shouldStmt | whilstStmt | block 

printStmt   ::= "print" equality ";"
makeStmt    ::= "make" expression ";"
exprStmt    ::= expression ";"
shouldStmt  ::= "should" "(" equality ")" block ("then" block)?
whilstStmt  ::= "whilst" "(" equality ")" block
block       ::= "{" statement* "}"

expression  ::= assignment
assignment  ::= IDENTIFIER "be" assignment | equality

equality    ::= comparison (("==" | "!=") comparison)?
comparison  ::= term ((<" | ">" | "<=" | ">=") term)?
term        ::= factor (("+" | "-") factor)*
factor      ::= unary (("*" | "/") unary)*
unary       ::= ("!" | "-") unary | primary
primary     ::= NUMBER | STRING | IDENTIFIER | BOOL | "(" expression ")"
